---TABLA PERFIL---

create table profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text,
  avatar_url text,
  theme text default 'midnight',
  quick_notes text default '',
  created_at timestamptz default now(),
  onboarding_completed boolean default false
);

---TABLA TAREAS---

create table tasks (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id) on delete cascade,

  title text not null,
  description text,

  status text default 'pending', -- pending | done

  position float8 not null, -- ORDEN MANUAL

  priority int default 2, -- 1 alta | 2 media | 3 baja

  estimated_minutes int,
  progress int default 0,

  total_focus_time int default 0, -- acumulado (mins),
  is_running boolean default false,

  due_date timestamptz,
  start_date timestamptz,

  pomodoro_cycles int default 4,
  pomodoro_duration int default 25,
  short_break_duration int default 5,
  long_break_duration int default 15,
  completed_cycles int4 default 0,

  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

---SESIONES DE POMODORO---

create table pomodoro_sessions (
  id uuid primary key default gen_random_uuid(),
  task_id uuid references tasks(id) on delete cascade,
  user_id uuid references auth.users(id) on delete cascade,

  duration_minutes int not null,
  started_at timestamptz,
  completed_at timestamptz,

  created_at timestamptz default now()
);

---NOTIFICACIONES---

create table reminders (
  id uuid primary key default gen_random_uuid(),
  task_id uuid references tasks(id) on delete cascade drop not null,
  user_id uuid references auth.users(id) on delete cascade,

  remind_at timestamptz not null,
  message text,
  sent boolean default false,

  created_at timestamptz default now()
);

---ANOTACIONES---
create table task_notes (
  id uuid primary key default gen_random_uuid(),
  task_id uuid references tasks(id) on delete cascade,
  user_id uuid references auth.users(id) on delete cascade,
  position float8 not null,

  content text not null,
  created_at timestamptz default now()
);

---ETIQUETAS---
create table tags (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,

  name text not null,
  color text,

  created_at timestamptz default now(),

  unique (user_id, name)
);

---ETIQUETAS TAREAS---
create table task_tags (
  task_id uuid not null references tasks(id) on delete cascade,
  tag_id uuid not null references tags(id) on delete cascade,

  primary key (task_id, tag_id)
);

alter table task_tags enable row level security;

create policy "Users manage their task tags"
on task_tags
for all
using (
  exists (
    select 1
    from tasks
    where tasks.id = task_tags.task_id
      and tasks.user_id = auth.uid()
  )
)
with check (
  exists (
    select 1
    from tasks
  )
);

--- TABLEROS COMPARTIDOS (SHARED LINKS) ---
CREATE TABLE IF NOT EXISTS public.shared_links (
    id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
    user_id UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    token TEXT NOT NULL UNIQUE,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    UNIQUE(user_id) -- Sólo un enlace activo por usuario
);

-- RLS para shared_links
ALTER TABLE public.shared_links ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view their own shared link" 
    ON public.shared_links FOR SELECT 
    USING (auth.uid() = user_id);

CREATE POLICY "Users can insert their own shared link" 
    ON public.shared_links FOR INSERT 
    WITH CHECK (auth.uid() = user_id);

CREATE POLICY "Users can update their own shared link" 
    ON public.shared_links FOR UPDATE 
    USING (auth.uid() = user_id);

-- RPC function para obtener datos del tablero sin exponer RLS
CREATE OR REPLACE FUNCTION get_shared_board_data(p_token TEXT)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER -- Permite ejecutar la función con permisos de administrador (bypasseando RLS para la vista pública)
AS $$
DECLARE
    v_user_id UUID;
    v_profile JSONB;
    v_tasks JSONB;
    v_tags JSONB;
    v_task_tags JSONB;
    v_result JSONB;
BEGIN
    -- 1. Verify token and get user_id
    SELECT user_id INTO v_user_id
    FROM public.shared_links
    WHERE token = p_token AND is_active = true;

    IF v_user_id IS NULL THEN
        RETURN NULL;
    END IF;

    -- 2. Get Profile info
    SELECT to_jsonb(p) INTO v_profile
    FROM public.profiles p
    WHERE id = v_user_id;

    -- 3. Get Tasks
    SELECT COALESCE(jsonb_agg(t), '[]'::jsonb) INTO v_tasks
    FROM public.tasks t
    WHERE user_id = v_user_id;

    -- 4. Get Tags
    SELECT COALESCE(jsonb_agg(tg), '[]'::jsonb) INTO v_tags
    FROM public.tags tg
    WHERE user_id = v_user_id;

    -- 5. Get Task Tags
    SELECT COALESCE(jsonb_agg(tt), '[]'::jsonb) INTO v_task_tags
    FROM public.task_tags tt
    WHERE task_id IN (SELECT id FROM public.tasks WHERE user_id = v_user_id);

    -- 6. Combine all into a single JSONB object
    v_result := jsonb_build_object(
        'profile', v_profile,
        'tasks', v_tasks,
        'tags', v_tags,
        'task_tags', v_task_tags
    );

    RETURN v_result;
END;
$$;
